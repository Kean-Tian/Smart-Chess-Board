cmake_minimum_required(VERSION 3.16)
set(IDF_TARGET "esp32s2" CACHE STRING "ESP32-S2-Saola-1")
set(EXTRA_COMPONENT_DIRS "${CMAKE_CURRENT_LIST_DIR}/main")
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
set(COMPONENTS main)
project(scb_s2_saola)
if(NOT IDF_TARGET STREQUAL "esp32s2")
    message(FATAL_ERROR "Use a clean build with idf.py set-target esp32s2 for ESP32-S2-Saola-1")
endif()
if("${IDF_VERSION_MAJOR}.${IDF_VERSION_MINOR}.${IDF_VERSION_PATCH}" VERSION_LESS "5.5.3")
    message(FATAL_ERROR "Use ESP-IDF 5.5.3 or newer; this project is validated with 5.5.3")
endif()
idf_build_get_property(scb_elf EXECUTABLE)
find_program(scb_size_tool NAMES xtensa-esp32s2-elf-size xtensa-esp-elf-size)
if(NOT scb_size_tool)
    message(FATAL_ERROR "Activate the ESP-IDF Xtensa toolchain before building")
endif()
add_custom_command(TARGET ${scb_elf} POST_BUILD
    COMMAND ${PYTHON} "${CMAKE_CURRENT_LIST_DIR}/scripts/ram_budget.py"
        --elf "$<TARGET_FILE:${scb_elf}>" --size-tool "${scb_size_tool}"
    VERBATIM)
