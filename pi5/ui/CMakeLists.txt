cmake_minimum_required(VERSION 3.20)
project(SCB_Touch_UI VERSION 0.1.0 LANGUAGES C CXX)

set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
option(SCB_UI_HEADLESS "Build the screenshot/test renderer without SDL2" OFF)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
include(FetchContent)
find_package(SQLite3 REQUIRED)
# One schema is shared by the C player manager and the Python game service.
set(SCB_SCHEMA_FILE "${CMAKE_CURRENT_SOURCE_DIR}/../storage/schema.sql")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${SCB_SCHEMA_FILE}")
file(READ "${SCB_SCHEMA_FILE}" SCB_SCHEMA_SQL)
string(REPLACE "\\" "\\\\" SCB_SCHEMA_SQL "${SCB_SCHEMA_SQL}")
string(REPLACE "\"" "\\\"" SCB_SCHEMA_SQL "${SCB_SCHEMA_SQL}")
string(REPLACE "\n" "\\n\"\n\"" SCB_SCHEMA_SQL "${SCB_SCHEMA_SQL}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/storage_schema.h" "static const char scb_schema_sql[] = \"${SCB_SCHEMA_SQL}\";\n")
# Pin both the release and checksum so a later upstream release cannot change the UI API.
FetchContent_Declare(lvgl
    URL https://codeload.github.com/lvgl/lvgl/tar.gz/refs/tags/v9.3.0
    URL_HASH SHA256=4933becfd3603b29158a5d04138139582836ef2bc17beb6c39dccda9cb0d32e7)
FetchContent_MakeAvailable(lvgl)

add_library(scb_players STATIC src/player_store.c)
target_include_directories(scb_players PUBLIC src)
target_include_directories(scb_players PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
if(TARGET SQLite3::SQLite3)
    target_link_libraries(scb_players PUBLIC SQLite3::SQLite3)
else()
    target_link_libraries(scb_players PUBLIC SQLite::SQLite3)
endif()
target_compile_options(scb_players PRIVATE -Wall -Wextra -Werror)
add_executable(scb_touch_ui src/main.c src/scb_ui.c src/ui_bridge.c)
target_include_directories(scb_touch_ui PRIVATE src)
target_link_libraries(scb_touch_ui PRIVATE lvgl::lvgl scb_players m)
target_compile_definitions(scb_touch_ui PRIVATE
    _POSIX_C_SOURCE=200809L
    SCB_UI_ASSET_DIR="${CMAKE_CURRENT_SOURCE_DIR}/assets")
target_compile_options(scb_touch_ui PRIVATE -Wall -Wextra -Werror)
if(SCB_UI_HEADLESS)
    target_compile_definitions(scb_touch_ui PRIVATE SCB_UI_HEADLESS=1)
else()
    find_package(SDL2 CONFIG REQUIRED)
    target_link_libraries(scb_touch_ui PRIVATE SDL2::SDL2)
endif()
enable_testing()
add_executable(player_store_tests tests/player_store_tests.c)
target_link_libraries(player_store_tests PRIVATE scb_players)
target_compile_definitions(player_store_tests PRIVATE _POSIX_C_SOURCE=200809L)
if(APPLE)
    target_compile_definitions(scb_touch_ui PRIVATE _DARWIN_C_SOURCE=1)
    target_compile_definitions(player_store_tests PRIVATE _DARWIN_C_SOURCE=1)
endif()
target_compile_options(player_store_tests PRIVATE -Wall -Wextra -Werror)
add_test(NAME player_database COMMAND player_store_tests)
add_test(NAME touch_ui_interactions COMMAND scb_touch_ui --smoke-test)
if(NOT DEFINED Python3_EXECUTABLE AND EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../.venv/bin/python")
    set(Python3_EXECUTABLE "${CMAKE_CURRENT_SOURCE_DIR}/../.venv/bin/python")
endif()
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
    add_test(NAME touch_ui_service_integration COMMAND "${Python3_EXECUTABLE}" -B
        "${CMAKE_CURRENT_SOURCE_DIR}/../tests/ui_service_tests.py")
    set_tests_properties(touch_ui_service_integration PROPERTIES
        ENVIRONMENT "SCB_TOUCH_UI_BINARY=$<TARGET_FILE:scb_touch_ui>" TIMEOUT 60)
endif()
if(NOT SCB_UI_HEADLESS)
    add_test(NAME touch_ui_pointer_input COMMAND scb_touch_ui --input-test)
    set_tests_properties(touch_ui_pointer_input PROPERTIES
        ENVIRONMENT "SDL_VIDEODRIVER=dummy;SDL_AUDIODRIVER=dummy")
endif()
